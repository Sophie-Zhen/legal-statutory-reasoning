%% BEGIN_FILE: statute_152.pl
:- module(statute_152,
          [ is_dependent/4,
            is_qualifying_child/3,
            qualifying_child_e/3,
            relationship_d2/3
          ]).

/**
 * statute_152.pl
 *
 * This module implements §152, defining who qualifies as a "dependent".
 * A dependent is either a "qualifying child" or a "qualifying relative".
 * Each category has its own set of specific tests.
 *
 * This module exports the main predicate is_dependent/4, as well as helpers
 * for testing specific sub-conditions required by the test cases.
 */

:- use_module(helpers).

% The system will provide the 'fact' predicates at runtime.
:- discontiguous fact/1, fact/2, fact/3, fact/4.

% is_dependent(+Taxpayer, +Individual, +Year, -Category)
%
% Top-level predicate to determine if an Individual is a dependent of a Taxpayer.
% Succeeds if the Individual is either a qualifying child or a qualifying relative,
% and none of the exceptions in §152(b) apply.
% Category will be unified with 'qualifying_child' or 'qualifying_relative'.
is_dependent(Taxpayer, Individual, Year, Category) :-
    \+ is_exception_b(Individual, Year),
    (
        is_qualifying_child(Taxpayer, Individual, Year),
        Category = qualifying_child
    ;
        \+ is_qualifying_child_of_any_taxpayer(Individual, Year),
        is_qualifying_relative(Taxpayer, Individual, Year),
        Category = qualifying_relative
    ).

% is_exception_b(+Individual, +Year)
%
% Implements the exceptions in §152(b).
% (1) A dependent cannot have their own dependents.
% (2) An individual who files a joint return with their spouse cannot be a dependent.
is_exception_b(Individual, Year) :-
    fact(is_dependent_of(Individual, _, Year)). % §152(b)(1)
is_exception_b(Individual, Year) :-
    fact(files_joint_return(Individual, Year)). % §152(b)(2)

% is_qualifying_child(+Taxpayer, +Individual, +Year)
%
% Implements §152(c)(1). An individual is a qualifying child if they meet
% all the tests: relationship, abode, age, and joint return.
is_qualifying_child(Taxpayer, Individual, Year) :-
    qualifying_child_relationship(Taxpayer, Individual),        % (A)
    fact(has_same_principal_abode(Individual, Taxpayer, Year, more_than_half)), % (B)
    meets_age_requirements_c3(Taxpayer, Individual, Year),      % (C)
    qualifying_child_e(Individual, Taxpayer, Year).              % (E)

% is_qualifying_child_of_any_taxpayer(+Individual, +Year)
%
% Checks if an individual is a qualifying child of ANY taxpayer. This is
% needed for the qualifying relative test in §152(d)(1)(D).
is_qualifying_child_of_any_taxpayer(Individual, Year) :-
    fact(is_potential_taxpayer(AnyTaxpayer)),
    AnyTaxpayer \== Individual,
    is_qualifying_child(AnyTaxpayer, Individual, Year).

% qualifying_child_relationship(+Taxpayer, +Individual)
%
% Implements §152(c)(2). Checks the relationship test for a qualifying child.
qualifying_child_relationship(Taxpayer, Individual) :- fact(child_of(Individual, Taxpayer)).
qualifying_child_relationship(Taxpayer, Individual) :- fact(child_of(Child, Taxpayer)), fact(descendant_of(Individual, Child)).
qualifying_child_relationship(Taxpayer, Individual) :- fact(sibling_of(Individual, Taxpayer)).
qualifying_child_relationship(Taxpayer, Individual) :- fact(step_sibling_of(Individual, Taxpayer)).
qualifying_child_relationship(Taxpayer, Individual) :- fact(sibling_of(Sibling, Taxpayer)), fact(descendant_of(Individual, Sibling)).

% meets_age_requirements_c3(+Taxpayer, +Individual, +Year)
%
% Implements §152(c)(3). The individual must be younger than the taxpayer
% and under 25 at the end of the year.
meets_age_requirements_c3(Taxpayer, Individual, Year) :-
    fact(date_of_birth(Individual, IY, IM, _ID)),
    fact(date_of_birth(Taxpayer, TY, TM, _TD)),
    helpers:get_age_at_year_end(IY, IM, Year, AgeOfIndividual),
    helpers:get_age_at_year_end(TY, TM, Year, AgeOfTaxpayer),
    AgeOfIndividual < AgeOfTaxpayer,
    AgeOfIndividual < 25.

% qualifying_child_e(+Individual, +Taxpayer, +Year)
%
% Implements §152(c)(1)(E). The individual has not filed a joint return.
% Exported for test case s152_c_1_E_pos. The Taxpayer argument is included for
% signature consistency, but is not used.
qualifying_child_e(Individual, _Taxpayer, Year) :-
    \+ (fact(files_joint_return(Individual, Year)), \+ fact(joint_return_for_refund_only(Individual, Year))).

% is_qualifying_relative(+Taxpayer, +Individual, +Year)
%
% Implements §152(d)(1). An individual is a qualifying relative if they meet
% all the tests: relationship, income, and not being a qualifying child of anyone.
is_qualifying_relative(Taxpayer, Individual, Year) :-
    qualifying_relative_relationship(Taxpayer, Individual, Year),     % (A)
    fact(has_no_income(Individual, Year)),                          % (B)
    \+ is_qualifying_child_of_any_taxpayer(Individual, Year).        % (D)

% qualifying_relative_relationship(+Taxpayer, +Individual, +Year)
%
% Implements §152(d)(2). Checks if the individual has one of the specified
% relationships to the taxpayer. The Year is passed for relationship (H).
qualifying_relative_relationship(Taxpayer, Individual, Year) :-
    relationship_d2(Individual, Taxpayer, RelationshipType),
    % Ensure rule H is only valid for the specific year
    ( RelationshipType == 'H' ->
        \+ fact(is_spouse_at_any_time(Individual, Taxpayer, Year)),
        fact(has_same_principal_abode(Individual, Taxpayer, Year, full_year)),
        fact(is_member_of_household(Individual, Taxpayer, Year))
    ; true
    ).

% relationship_d2(+Individual, +Taxpayer, -RelationshipType)
%
% Implements the specific relationship clauses of §152(d)(2).
% Exported to satisfy test case s152_d_2_F_pos.
relationship_d2(Individual, Taxpayer, 'A') :- fact(child_of(Individual, Taxpayer)).
relationship_d2(Individual, Taxpayer, 'A') :- fact(child_of(Child, Taxpayer)), fact(descendant_of(Individual, Child)).
relationship_d2(Individual, Taxpayer, 'B') :- fact(sibling_of(Individual, Taxpayer)).
relationship_d2(Individual, Taxpayer, 'B') :- fact(step_sibling_of(Individual, Taxpayer)).
relationship_d2(Individual, Taxpayer, 'C') :- fact(parent_of(Individual, Taxpayer)).
relationship_d2(Individual, Taxpayer, 'C') :- fact(parent_of(Parent, Taxpayer)), fact(ancestor_of(Individual, Parent)).
relationship_d2(Individual, Taxpayer, 'D') :- fact(step_parent_of(Individual, Taxpayer)).
relationship_d2(Individual, Taxpayer, 'E') :- fact(sibling_of(Sibling, Taxpayer)), fact(child_of(Individual, Sibling)).
relationship_d2(Individual, Taxpayer, 'F') :- fact(parent_of(Parent, Taxpayer)), fact(sibling_of(Individual, Parent)).
relationship_d2(Individual, Taxpayer, 'G') :- fact(in_law_of(Individual, Taxpayer)).
relationship_d2(_Individual, _Taxpayer, 'H'). % Placeholder for type H check inside the caller.
