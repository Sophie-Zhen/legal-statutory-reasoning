% that need to be limited by section 68.

    fact(CaseID, deduction_choice(Taxpayer, TaxYear, 'itemize')),
    fact(CaseID, itemized_deductions_initial(Taxpayer, TaxYear, InitialDeductions)),
    !,
    fact(CaseID, adjusted_gross_income(Taxpayer, TaxYear, AGI)),
    section68:itemized_deduction_limitation(CaseID, Taxpayer, TaxYear, InitialDeductions, FinalDeductions),
    personal_exemption_deduction(CaseID, Taxpayer, TaxYear, PED),
    TaxableIncome is max(0, AGI - FinalDeductions - PED).
