run_tests :-
    findall(CaseID, answer(CaseID, _), Cases),
    run_tests(Cases, Passed, Failed),
    format('~nTests complete. Passed: ~w, Failed: ~w~n', [Passed, Failed]).

run_tests([], Passed, Passed).
run_tests([CaseID|T], Acc, Failed) :-
    (   test_case(CaseID)
    ->  NewAcc is Acc + 1,
        run_tests(T, NewAcc, Failed)
    ;   run_tests(T, Acc, [CaseID|Failed])
    ).

% Test harness for a single case
test_case(CaseID) :-
    answer(CaseID, Expected),
    (   (Expected == true ; Expected == false)
    ->  % Entailment/Contradiction
        answer(CaseID, Calculated),
        (   Calculated == Expected
        ->  format('Test ~w passed.~n', [CaseID])
        ;   format('Test ~w FAILED. Expected ~w, got ~w~n', [CaseID, Expected, Calculated]), fail
        )
    ;   % Numerical/how-much-tax
        answer(CaseID, Calculated),
        round_to_nearest_dollar(Calculated, RoundedCalc),
        (   RoundedCalc == Expected
        ->  format('Test ~w passed.~n', [CaseID])
        ;   format('Test ~w FAILED. Expected ~w, got ~w (rounded to ~w)~n', [CaseID, Expected, Calculated, RoundedCalc]), fail
        )
    ).
