% is_dependent(+CaseID, +Taxpayer, +PotentialDependent, +TaxYear)
% §152(a) In general
is_dependent(CaseID, Taxpayer, PotentialDependent, TaxYear) :-
    Taxpayer \== PotentialDependent,
    \+ exception_b(CaseID, Taxpayer, PotentialDependent, TaxYear),
    (   is_qualifying_child(CaseID, Taxpayer, PotentialDependent, TaxYear)
    ;   is_qualifying_relative(CaseID, Taxpayer, PotentialDependent, TaxYear)
    ).

% exception_b(+CaseID, +Taxpayer, +PotentialDependent, +TaxYear)
% §152(b) Exceptions
exception_b(CaseID, _Taxpayer, PotentialDependent, TaxYear) :-
    % (1) Dependents ineligible: If an individual is a dependent of a taxpayer,
    % such individual shall be treated as having no dependents.
    % This means we check if the *taxpayer* is a dependent of someone else.
    is_dependent_of_another(CaseID, Taxpayer, TaxYear).
exception_b(CaseID, _Taxpayer, PotentialDependent, TaxYear) :-
    % (2) Married dependents: An individual shall not be treated as a dependent...
    % if such individual has made a joint return with the individual's spouse
    get_spouse(CaseID, PotentialDependent, Spouse),
    fact(CaseID, filing_status(PotentialDependent, TaxYear, 'joint')),
    fact(CaseID, files_joint_return_with(PotentialDependent, Spouse, TaxYear)).

% A helper to check if a person is a dependent of ANY taxpayer in the knowledge base.
is_dependent_of_another(CaseID, Person, TaxYear) :-
    fact(CaseID, taxpayer(OtherTaxpayer, TaxYear)),
    OtherTaxpayer \== Person,
    is_dependent(CaseID, OtherTaxpayer, Person, TaxYear).

% count_exemptions(+CaseID, +Taxpayer, +TaxYear, -Count)
count_exemptions(CaseID, Taxpayer, TaxYear, Count) :-

exemption_for_taxpayer(CaseID, Taxpayer, TaxYear, 1) :-

% exemption_for_spouse(+CaseID, +Taxpayer, +TaxYear, -Count)
% §151(b)
exemption_for_spouse(CaseID, Taxpayer, TaxYear, 1) :-
    helpers:get_spouse(CaseID, Taxpayer, Spouse),
    % if a joint return is not made
    (fact(CaseID, filing_status(Taxpayer, TaxYear, Status)) -> Status \= 'joint' ; true),
    % spouse has no gross income
    (fact(CaseID, gross_income(Spouse, TaxYear, GI)) -> GI =:= 0 ; fact(CaseID, no_income(Spouse, TaxYear))),
    % spouse is not the dependent of another taxpayer
    \+ can_be_claimed_as_dependent(CaseID, Spouse, TaxYear),
    !.
exemption_for_spouse(_, _, _, 0).

% exemption_for_dependents(+CaseID, +Taxpayer, +TaxYear, -Count)

exemption_for_dependents(CaseID, Taxpayer, TaxYear, Count) :-
    findall(Dep, is_dependent(CaseID, Taxpayer, Dep, TaxYear), Dependents),
    length(Dependents, Count).

    !.
exemption_amount_per_person(CaseID, Taxpayer, TaxYear, PhasedOutAmount) :-
    personal_exemption_amount(TaxYear, BaseAmount),
    phaseout_reduction_percentage(CaseID, Taxpayer, TaxYear, ReductionPercentage),

% phaseout_reduction_percentage(+CaseID, +Taxpayer, +TaxYear, -Percentage)
