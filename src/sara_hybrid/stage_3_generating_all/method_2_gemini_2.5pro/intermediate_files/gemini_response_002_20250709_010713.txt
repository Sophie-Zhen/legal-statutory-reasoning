% get_age_at_year_end(+CaseID, +Person, +TaxYear, -Age)
% Calculates the age of a person at the end of a given tax year.
% "attained age 65 before the close of his taxable year" means the person's
% 65th birthday is on or before Dec 31 of the tax year.
get_age_at_year_end(CaseID, Person, TaxYear, Age) :-
    fact(CaseID, birth_date(Person, Y, M, D)),
    ( (M < 12 ; (M =:= 12, D =< 31)) -> Age is TaxYear - Y
    ; Age is TaxYear - Y - 1
    ).
% Handle cases where birth date is not available.
get_age_at_year_end(CaseID, Person, TaxYear, Age) :-
    \+ fact(CaseID, birth_date(Person, _, _, _)),
    fact(CaseID, born(Person, Y)), % Simplified fact for some cases
    Age is TaxYear - Y.

% calculate_tax_from_brackets(+TaxableIncome, +Brackets, -Tax)
% Calculates tax based on a given income and a list of tax brackets.
% Brackets must be a list of bracket(IncomeOver, Rate, BaseTax)
% sorted in descending order of IncomeOver.
calculate_tax_from_brackets(TaxableIncome, [bracket(Limit, Rate, BaseTax) | _], Tax) :-
    TaxableIncome > Limit,
    !,
    Excess is TaxableIncome - Limit,
    Tax is BaseTax + (Excess * Rate).
calculate_tax_from_brackets(TaxableIncome, [_ | Rest], Tax) :-
    calculate_tax_from_brackets(TaxableIncome, Rest, Tax).
calculate_tax_from_brackets(_, [], 0). % Base case for income below lowest bracket.

% get_spouse(+CaseID, +Person, -Spouse)
% Finds the spouse of a person in a given case context.
get_spouse(CaseID, Person, Spouse) :-
    ( fact(CaseID, married(Person, Spouse, _))
    ; fact(CaseID, married(Spouse, Person, _))
    ),
    Person \== Spouse.
