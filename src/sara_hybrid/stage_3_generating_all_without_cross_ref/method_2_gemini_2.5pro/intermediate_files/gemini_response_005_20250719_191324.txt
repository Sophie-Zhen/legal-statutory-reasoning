/** <module> section152
 *
 * Encodes §152, which defines who qualifies as a "dependent", broken
 * down into "qualifying child" and "qualifying relative".
 */

% §152(a) Dependent defined
is_dependent(CaseID, Taxpayer, PotentialDependent, Year) :-
    \+ is_dependent_exception(CaseID, PotentialDependent, Year),
    (   is_qualifying_child(CaseID, Taxpayer, PotentialDependent, Year)
    ;   is_qualifying_relative(CaseID, Taxpayer, PotentialDependent, Year)
    ).

% §152(b) Exceptions
is_dependent_exception(CaseID, Person, Year) :-
    % (1) Dependent cannot have their own dependents
    fact(CaseID, potential_dependent(Person, DepOfDep)),
    is_dependent(CaseID, Person, DepOfDep, Year),
    !.
is_dependent_exception(CaseID, Person, Year) :-
    % (2) Married dependents filing jointly
    fact(CaseID, spouse(Person, Spouse)),
    fact(CaseID, filing_status(Person, married_filing_jointly, Year)),
    fact(CaseID, filing_status(Spouse, married_filing_jointly, Year)).

% §152(c) Qualifying child
is_qualifying_child(CaseID, Taxpayer, PotentialChild, Year) :-
    relationship_qualifying_child(CaseID, Taxpayer, PotentialChild),
    abode_test(CaseID, Taxpayer, PotentialChild, Year),
    age_test(CaseID, Taxpayer, PotentialChild, Year),
    qualifying_child_joint_return_test(CaseID, PotentialChild, Year).

% §152(c)(2) Relationship test for QC
relationship_qualifying_child(CaseID, Taxpayer, P) :- fact(CaseID, relationship(P, Taxpayer, R)), member(R, [child, son, daughter, stepchild, stepson, stepdaughter, descendant_of_child, brother, sister, stepbrother, stepsister, descendant_of_sibling]).

% §152(c)(1)(B) Abode test for QC
abode_test(CaseID, Taxpayer, PotentialChild, Year) :-
    fact(CaseID, principal_place_of_abode(PotentialChild, Abode, Year)),
    fact(CaseID, principal_place_of_abode(Taxpayer, Abode, Year)),
    fact(CaseID, lived_together_more_than_half_year(PotentialChild, Taxpayer, Year)).

% §152(c)(3) Age test for QC
age_test(CaseID, Taxpayer, PotentialChild, Year) :-
    get_age_at_year_end(CaseID, PotentialChild, Year, ChildAge),
    get_age_at_year_end(CaseID, Taxpayer, Year, TaxpayerAge),
    ChildAge < TaxpayerAge,
    ChildAge < 25. % Simplified from the statute's complex student/disability rules

% §152(c)(1)(E) Joint return test for QC
qualifying_child_joint_return_test(CaseID, PotentialChild, Year) :-
    (   fact(CaseID, spouse(PotentialChild, Spouse))
    ->  \+ (fact(CaseID, filing_status(PotentialChild, married_filing_jointly, Year)), fact(CaseID, filing_status(Spouse, married_filing_jointly, Year)))
    ;   true
    ).

% §152(d) Qualifying relative
is_qualifying_relative(CaseID, Taxpayer, PotentialRelative, Year) :-
    relationship_qualifying_relative(CaseID, Taxpayer, PotentialRelative, Year),
    income_test_qualifying_relative(CaseID, PotentialRelative, Year),
    not_qualifying_child_test(CaseID, Taxpayer, PotentialRelative, Year).

% §152(d)(2) Relationship test for QR
relationship_qualifying_relative(CaseID, Taxpayer, P, Year) :-
    relationship_qualifying_relative_by_blood_or_marriage(CaseID, Taxpayer, P) ;
    relationship_qualifying_relative_household_member(CaseID, Taxpayer, P, Year).

relationship_qualifying_relative_by_blood_or_marriage(CaseID, Taxpayer, P) :-
    fact(CaseID, relationship(P, Taxpayer, R)),
    member(R, [child, descendant_of_child, brother, sister, stepbrother, stepsister,
               father, mother, ancestor_of_parent, stepfather, stepmother,
               son_of_sibling, daughter_of_sibling, uncle, aunt,
               son_in_law, daughter_in_law, father_in_law, mother_in_law,
               brother_in_law, sister_in_law]).

% §152(d)(2)(H) Household member test
relationship_qualifying_relative_household_member(CaseID, Taxpayer, P, Year) :-
    fact(CaseID, principal_place_of_abode(P, Abode, Year)),
    fact(CaseID, principal_place_of_abode(Taxpayer, Abode, Year)),
    \+ (fact(CaseID, spouse(P, Taxpayer)); fact(CaseID, spouse(Taxpayer, P))),
    \+ fact(CaseID, former_spouse(P, Taxpayer)).

% §152(d)(1)(B) Income test for QR
income_test_qualifying_relative(CaseID, PotentialRelative, Year) :-
    \+ (fact(CaseID, gross_income(PotentialRelative, Year, GI)), GI > 0).

% §152(d)(1)(D) Not a qualifying child test for QR
not_qualifying_child_test(CaseID, Taxpayer, PotentialRelative, Year) :-

    \+ (fact(CaseID, potential_claimer(OtherClaimer)), OtherClaimer \== Taxpayer, is_qualifying_child(CaseID, OtherClaimer, PotentialRelative, Year)).
