% is_surviving_spouse(+CaseID, +Taxpayer, +TaxYear)
% §2(a) Definition of surviving spouse
is_surviving_spouse(CaseID, Taxpayer, TaxYear) :-
    % (1)(A) spouse died during either of the two years immediately preceding
    get_spouse(CaseID, Taxpayer, Spouse),
    fact(CaseID, death_of(Spouse, DeathDate)),
    fact(CaseID, date(DeathDate, DeathYear, _, _)),
    TaxYear - DeathYear >= 1,
    TaxYear - DeathYear =< 2,
    % (1)(B) maintains home for dependent child
    fact(CaseID, maintains_household(Taxpayer, TaxYear)),
    fact(CaseID, principal_place_of_abode(Child, Taxpayer, TaxYear)),
    fact(CaseID, relationship(Child, Taxpayer, R)), member(R, [son, stepson, daughter, stepdaughter]),
    is_dependent(CaseID, Taxpayer, Child, TaxYear),
    personal_exemption_deduction_for_dependent(CaseID, Taxpayer, Child, TaxYear),
    % (2) Limitations
    \+ has_remarried(CaseID, Taxpayer, TaxYear),

% has_remarried(+CaseID, +Taxpayer, +TaxYear)
has_remarried(CaseID, Taxpayer, TaxYear) :-

% maintains_home_for_qualifying_person_A(+CaseID, +Taxpayer, +TaxYear)

maintains_home_for_qualifying_person_A(CaseID, Taxpayer, TaxYear) :-
    fact(CaseID, furnishes_over_half_cost_of_household(Taxpayer, TaxYear)),
    (
        % (i) a qualifying child...
        fact(CaseID, principal_place_of_abode_over_half_year(Child, Taxpayer, TaxYear)),
        is_qualifying_child(CaseID, Taxpayer, Child, TaxYear),
        \+ (is_married(CaseID, Child, TaxYear), \+ is_dependent(CaseID, Taxpayer, Child, TaxYear)) % not married and not a dependent
        ;

        fact(CaseID, principal_place_of_abode_over_half_year(OtherPerson, Taxpayer, TaxYear)),
        is_dependent(CaseID, Taxpayer, OtherPerson, TaxYear),
        personal_exemption_deduction_for_dependent(CaseID, Taxpayer, OtherPerson, TaxYear),
        % (3)(B) limitation on 152(d)(2)(H)

        \+ fact(CaseID, member_of_household_for_year(Dependent, Taxpayer, _))
