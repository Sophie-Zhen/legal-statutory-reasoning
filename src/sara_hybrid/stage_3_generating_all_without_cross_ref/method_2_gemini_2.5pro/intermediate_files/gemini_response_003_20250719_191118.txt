% §2(a) Definition of surviving spouse
is_surviving_spouse(CaseID, Taxpayer, Year) :-
    is_surviving_spouse_requirements(CaseID, Taxpayer, Year),
    \+ is_surviving_spouse_limitation(CaseID, Taxpayer, Year).

is_surviving_spouse_requirements(CaseID, Taxpayer, Year) :-
    fact(CaseID, spouse(Taxpayer, Spouse)),
    fact(CaseID, death_date(Spouse, date(DeathYear, _, _))),
    PrecedingYear1 is Year - 1,
    PrecedingYear2 is Year - 2,
    (DeathYear =:= PrecedingYear1 ; DeathYear =:= PrecedingYear2),
    maintains_household_for_dependent_child(CaseID, Taxpayer, Year).

maintains_household_for_dependent_child(CaseID, Taxpayer, Year) :-
    is_dependent(CaseID, Taxpayer, Dependent, Year),
    (fact(CaseID, relationship(Dependent, Taxpayer, son)) ; fact(CaseID, relationship(Dependent, Taxpayer, daughter))),
    is_deduction_for_dependent_allowed(CaseID, Taxpayer, Dependent, Year),
    fact(CaseID, principal_place_of_abode(Dependent, Location, Year)),
    fact(CaseID, maintains_household(Taxpayer, Location, Year)),
    fact(CaseID, cost_of_maintaining_household(Taxpayer, Year, PayerCost)),
    fact(CaseID, cost_of_maintaining_household(total, Year, TotalCost)),
    PayerCost > TotalCost / 2.

is_surviving_spouse_limitation(CaseID, Taxpayer, Year) :-
    is_married(CaseID, Taxpayer, Year) ;
    (   fact(CaseID, spouse(Taxpayer, DeceasedSpouse)),
        fact(CaseID, death_date(DeceasedSpouse, date(DeathYear, _, _))),
        \+ surviving_spouse_joint_return_could_be_made(CaseID, Taxpayer, DeathYear)
    ).

% §2(a)(2)(B) - check if a joint return could have been made
surviving_spouse_joint_return_could_be_made(CaseID, Taxpayer, Year) :-
    fact(CaseID, spouse(Taxpayer, Spouse)),
    \+ fact(CaseID, nonresident_alien(Taxpayer, Year)),
    \+ fact(CaseID, nonresident_alien(Spouse, Year)).

% §2(b) Definition of head of household
is_head_of_household(CaseID, Taxpayer, Year) :-
    is_head_of_household_requirements(CaseID, Taxpayer, Year),
    \+ is_head_of_household_limitation(CaseID, Taxpayer, Year).

is_head_of_household_requirements(CaseID, Taxpayer, Year) :-
    \+ is_married(CaseID, Taxpayer, Year),
    \+ is_surviving_spouse(CaseID, Taxpayer, Year),
    (   maintains_household_for_qualifying_person(CaseID, Taxpayer, Year)
    ;   maintains_household_for_parent(CaseID, Taxpayer, Year)
    ).

maintains_household_for_qualifying_person(CaseID, Taxpayer, Year) :-
    (is_qualifying_child(CaseID, Taxpayer, Child, Year) -> QualifyingPerson = Child ; is_dependent(CaseID, Taxpayer, Other, Year), QualifyingPerson = Other),
    fact(CaseID, principal_place_of_abode_more_than_half_year(QualifyingPerson, Location, Year)),
    fact(CaseID, maintains_household(Taxpayer, Location, Year)),
    fact(CaseID, cost_of_maintaining_household(Taxpayer, Year, PayerCost)),
    fact(CaseID, cost_of_maintaining_household(total, Year, TotalCost)),
    PayerCost > TotalCost / 2.

maintains_household_for_parent(CaseID, Taxpayer, Year) :-
    is_dependent(CaseID, Taxpayer, Parent, Year),
    is_deduction_for_dependent_allowed(CaseID, Taxpayer, Parent, Year),
    fact(CaseID, parent(Parent, Taxpayer)),
    fact(CaseID, principal_place_of_abode(Parent, Location, Year)),
    fact(CaseID, maintains_household(Taxpayer, Location, Year)).

is_head_of_household_limitation(CaseID, Taxpayer, Year) :-
    fact(CaseID, nonresident_alien(Taxpayer, Year));
    (   is_dependent(CaseID, Taxpayer, Dependent, Year),
        \+ relationship_qualifying_relative(CaseID, Taxpayer, Dependent),
        relationship_qualifying_relative_household_member(CaseID, Taxpayer, Dependent, Year)
    ).

% §63(a) & (b) Taxable income defined
taxable_income(CaseID, Payer, Year, TaxableIncome) :-
    fact(CaseID, gross_income(Payer, Year, GrossIncome)),
    % AGI is assumed to be GrossIncome unless specified.
    (   fact(CaseID, takes_standard_deduction(Payer, Year))
    ->  standard_deduction(CaseID, Payer, Year, StandardDeduction),
        personal_exemption_deduction(CaseID, Payer, Year, PersonalExemptions),
        TaxableIncome is max(0, GrossIncome - StandardDeduction - PersonalExemptions)
    ;   itemized_deductions(CaseID, Payer, Year, FinalItemizedDeductions),
        personal_exemption_deduction(CaseID, Payer, Year, PersonalExemptions),
        TaxableIncome is max(0, GrossIncome - FinalItemizedDeductions - PersonalExemptions)
    ).

% §63(d) Itemized deductions
itemized_deductions(CaseID, Payer, Year, FinalDeductions) :-
    fact(CaseID, itemized_deductions(Payer, Year, InitialDeductions)),
    % This is where §68 limitation would be applied.
    % For now, we assume the provided value is final.
    FinalDeductions = InitialDeductions.

% §63(c) Standard deduction
standard_deduction(CaseID, Payer, Year, TotalStandardDeduction) :-
    is_eligible_for_standard_deduction(CaseID, Payer, Year), !,
    calculate_basic_standard_deduction(CaseID, Payer, Year, Basic),
    calculate_additional_standard_deduction(CaseID, Payer, Year, Additional),
    TotalStandardDeduction is Basic + Additional.
standard_deduction(_CaseID, _Payer, _Year, 0). % Not eligible

is_eligible_for_standard_deduction(CaseID, Payer, Year) :-
    \+ fact(CaseID, nonresident_alien(Payer, Year)),
    \+ ( is_married(CaseID, Payer, Year),
         fact(CaseID, filing_status(Payer, married_filing_separately, Year)),
         fact(CaseID, spouse(Payer, Spouse)),
         \+ fact(CaseID, takes_standard_deduction(Spouse, Year))
       ).

% §63(c)(2), (c)(5), (c)(7) Basic standard deduction

    ).

% §63(f), (c)(3) Additional standard deduction
calculate_additional_standard_deduction(CaseID, Payer, Year, TotalAdditional) :-
    get_additional_for_self(CaseID, Payer, Year, SelfAmount),

get_additional_for_self(CaseID, Payer, Year, Amount) :-
