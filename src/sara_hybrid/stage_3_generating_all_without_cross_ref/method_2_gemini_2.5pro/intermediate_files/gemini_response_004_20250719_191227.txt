% is_married/3(+CaseID, +Person, +Year)
% Determines if a person is considered married for the given tax year.
is_married(CaseID, Person, Year) :-
    % §7703(a) general rule
    (   fact(CaseID, married(Person, _, Year))
    ;   fact(CaseID, spouse(Person, Spouse)),
        fact(CaseID, death_date(Spouse, date(Year, _, _)))
    ),

    \+ is_considered_unmarried_living_apart(CaseID, Person, Year).

% is_considered_unmarried_living_apart/3(+CaseID, +Person, +Year)
% Implements the §7703(b) exception for certain married individuals living apart.
is_considered_unmarried_living_apart(CaseID, Person, Year) :-
    % Condition: Must be married under (a) to be considered for this exception.
    (fact(CaseID, married(Person, _, Year)) ; (fact(CaseID, spouse(Person, Spouse)), fact(CaseID, death_date(Spouse, date(Year, _, _))))),
    \+ fact(CaseID, legally_separated(Person, _, Year)),
    % (1) files a separate return
    fact(CaseID, filing_status(Person, married_filing_separately, Year)),
    % (1) maintains a household for a child
    maintains_home_for_child(CaseID, Person, Year),
    % (2) furnishes over half the cost of the household
    fact(CaseID, cost_of_maintaining_household(Person, Year, PayerCost)),
    fact(CaseID, cost_of_maintaining_household(total, Year, TotalCost)),
    PayerCost > TotalCost / 2,
    % (3) spouse is not a member of the household for the last 6 months
    fact(CaseID, spouse(Person, Spouse)),
    \+ fact(CaseID, member_of_household_last_6_months(Spouse, Year)).

maintains_home_for_child(CaseID, Person, Year) :-
    fact(CaseID, child(Child, Person)),
    fact(CaseID, principal_place_of_abode_more_than_half_year(Child, Location, Year)),
    fact(CaseID, maintains_household(Person, Location, Year)),
