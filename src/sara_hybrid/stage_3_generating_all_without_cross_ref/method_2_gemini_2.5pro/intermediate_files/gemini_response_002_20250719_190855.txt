:- use_module(library(lists), [sum_list/2]).

% Provides a safe way to get a fact, failing if not found.
get_fact(CaseID, Term, Value) :-
    fact(CaseID,..[Term, Value]).

% Determines if a given year falls under the Tax Cuts and Jobs Act (TCJA) general rules.
% §151(d)(5), §63(c)(7), §68(f) all reference years 2018 through 2025.
tcja_active_general(Year) :-
    integer(Year),
    Year >= 2018,
    Year =< 2025.

% Calculates age of a person at the end of a given year.
get_age_at_year_end(CaseID, Person, Year, Age) :-
    fact(CaseID, birth_date(Person, date(BirthYear, _, _))),
    Age is Year - BirthYear.

% Rounds a number to the nearest integer (dollar).
round_to_nearest_dollar(Number, Rounded) :-
    Rounded is round(Number).

% Calculates tax based on a progressive bracket system.
% Brackets is a list of [UpperBound, Rate, BaseTax].
calculate_tax_from_brackets(TaxableIncome, Brackets, Tax) :-
    calculate_tax_from_brackets(TaxableIncome, Brackets, 0, Tax).

calculate_tax_from_brackets(TaxableIncome, [[Cap, Rate, Base]|_], PrevCap, Tax) :-
    TaxableIncome > PrevCap,
    TaxableIncome =< Cap,
    !,
    Tax is Base + ((TaxableIncome - PrevCap) * Rate).
% Handle the top bracket (no upper cap).
calculate_tax_from_brackets(TaxableIncome, [[Cap, Rate, Base]], PrevCap, Tax) :-
    TaxableIncome > PrevCap,
    TaxableIncome > Cap, % This condition distinguishes it from the clause above
    !,
    Tax is Base + ((TaxableIncome - PrevCap) * Rate).
calculate_tax_from_brackets(TaxableIncome, [[Cap, _, _]|RestBrackets], _, Tax) :-
    calculate_tax_from_brackets(TaxableIncome, RestBrackets, Cap, Tax).
