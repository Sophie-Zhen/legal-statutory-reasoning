% is_married(+CaseID, +Person, +TaxYear)
% §7703 Determination of marital status
% Succeeds if Person is considered married for tax purposes in TaxYear.
is_married(CaseID, Person, TaxYear) :-
    % (a) General Rule
    % (2) An individual legally separated... shall not be considered as married.
    \+ fact(CaseID, legally_separated(Person, _, TaxYear)),
    (
        % (a)(1) ...determination... made as of the close of his taxable year
        fact(CaseID, married(Person, Spouse, StartDate)),
        fact(CaseID, date(StartDate, Y, _, _)),
        Y =< TaxYear,
        % Check if spouse died during the year
        (   get_spouse(CaseID, Person, Spouse),
            fact(CaseID, death_of(Spouse, DeathDate)),
            fact(CaseID, date(DeathDate, DeathYear, _, _)),
            DeathYear =:= TaxYear
        ->  true % Still married for the year of death
        ;   \+ (
                get_spouse(CaseID, Person, Spouse),
                fact(CaseID, death_of(Spouse, DeathDate)),
                fact(CaseID, date(DeathDate, DeathYear, _, _)),
                DeathYear < TaxYear
            )
        )
    ),
    % (b) Certain married individuals living apart shall not be considered as married.
    \+ is_considered_not_married_living_apart(CaseID, Person, TaxYear).

% is_considered_not_married_living_apart(+CaseID, +Person, +TaxYear)
% §7703(b)
% Succeeds if a person, while technically married, is treated as unmarried
% for certain provisions because they are living apart.
is_considered_not_married_living_apart(CaseID, Person, TaxYear) :-
    % Must be married under (a) to be considered for this subsection.
    fact(CaseID, married(Person, _, _)),
    \+ fact(CaseID, legally_separated(Person, _, TaxYear)),

    % (1) files a separate return
    fact(CaseID, filing_status(Person, TaxYear, 'married_filing_separately')),

    % (1) maintains as his home a household which constitutes for more than one-half of the
    % taxable year the principal place of abode of a child
    fact(CaseID, maintains_household_for_child(Person, Child, TaxYear)),
    fact(CaseID, lives_in_household(Child, Person, TaxYear, Duration)),
    Duration > 182, % More than half a year

    % (1) ...with respect to whom such individual is entitled to a deduction for the taxable year under section 151
    personal_exemption_deduction_for_dependent(CaseID, Person, Child, TaxYear),

    % (2) such individual furnishes over one-half of the cost of maintaining such household during the taxable year
    fact(CaseID, furnishes_over_half_cost_of_household(Person, TaxYear)),

    % (3) during the last 6 months of the taxable year, such individual's spouse is not a member of such household
    get_spouse(CaseID, Person, Spouse),
    \+ fact(CaseID, lives_in_household_last_6_months(Spouse, Person, TaxYear)).

% is_employer(+CaseID, +Person, +TaxYear)
% §3306(a)
is_employer(CaseID, Person, TaxYear) :-
    is_employer_general(CaseID, Person, TaxYear);
    is_employer_agricultural(CaseID, Person, TaxYear);
    is_employer_domestic(CaseID, Person, TaxYear).

% is_employer_general(+CaseID, +Person, +TaxYear)
% §3306(a)(1)
is_employer_general(CaseID, Person, TaxYear) :-
    \+ is_excluded_from_general_employer_test(CaseID, Person),
    (
        % Paid wages of $1,500 or more in current or preceding year
        futa_wage_threshold('general', TaxYear, WageLimit),
        (
            aggregate_all(sum(Wages), fact(CaseID, paid_wages(Person, _, Wages, TaxYear, 'general')), TotalWages), TotalWages >= WageLimit;
            PrevYear is TaxYear - 1,
            aggregate_all(sum(Wages), fact(CaseID, paid_wages(Person, _, Wages, PrevYear, 'general')), PrevTotalWages), PrevTotalWages >= WageLimit
        )
        ;
        % Employed at least one individual on each of some 10 days in different weeks
        futa_employee_threshold('general', TaxYear, EmployeeLimit),
        (
             findall(Week, fact(CaseID, employed_on_date(Person, _, _, Week, TaxYear, EmployeeLimit)), Weeks), length(Weeks, NumWeeks), NumWeeks >= 10;
             PrevYear is TaxYear - 1,
             findall(Week, fact(CaseID, employed_on_date(Person, _, _, Week, PrevYear, EmployeeLimit)), PrevWeeks), length(PrevWeeks, PrevNumWeeks), PrevNumWeeks >= 10
        )
    ).

% §3306(a)(1) ... there shall not be taken into account any wages paid to, or employment of, an employee performing domestic services
is_excluded_from_general_employer_test(CaseID, Person) :-
    fact(CaseID, service_type(Person, _, 'domestic')).

is_wage(CaseID, PaymentID, TaxYear) :-

is_wage_exception_limit(CaseID, PaymentID, TaxYear) :-

% futa_total_wages(+CaseID, +Employer, +TaxYear, -TotalWages)
% Calculates total taxable FUTA wages for an employer in a year.
futa_total_wages(CaseID, Employer, TaxYear, TotalWages) :-
    % Group payments by employee
    findall(Employee, fact(CaseID, payment(_, Employer, Employee, _, Date, _)), Employees),
    list_to_set(Employees, UniqueEmployees),
    % Calculate taxable wages for each employee and sum them up
    futa_wage_base_limit(TaxYear, Limit),
    aggregate_all(sum(TaxableAmount),
        (   member(E, UniqueEmployees),
            aggregate_all(sum(A), (
                fact(CaseID, payment(P, Employer, E, A, Date, _)),
                fact(CaseID, date(Date, TaxYear, _, _)),
                is_employment(CaseID, P, TaxYear),
                is_wage(CaseID, P, TaxYear)
            ), EmployeeTotalWages),
            TaxableAmount is min(EmployeeTotalWages, Limit)
        ),
    TotalWages).
