% Case ID: s152_d_2_F_pos
fact(s152_d_2_F_pos, person1_for_relationship(alice)). % Alice
fact(s152_d_2_F_pos, person2_for_relationship(bob)).   % Bob
fact(s152_d_2_F_pos, father_of(charlie, bob)). % Charlie is Bob's father
fact(s152_d_2_F_pos, sister_of(alice, charlie)). % Alice is Charlie's sister -> Alice is Bob's aunt
fact(s152_d_2_F_pos, question_type(entailment)).
fact(s152_d_2_F_pos, question_assertion_relationship(s152_d_2_F)). % Alice bears relationship to Bob

answer(s152_d_2_F_pos, Result) :-
    fact(s152_d_2_F_pos, person1_for_relationship(TaxpayerID)), % Alice
    fact(s152_d_2_F_pos, person2_for_relationship(IndividualID)), % Bob
    % s152(d)(2)(F): A brother or sister of the father or mother of the taxpayer.
    % Question: Alice bears a relationship to Bob under s152(d)(2)(F).
    % This means Bob is the "taxpayer" in the context of the statute's wording, Alice is the potential "relative".
    % So, IndividualID=Alice (potential relative), TaxpayerID=Bob (the reference person)
    section152:s152_d_2_relationship_qr(s152_d_2_F_pos, bob, alice, RelationshipMet), % Does Alice have a listed QR relationship to Bob?
    % Specifically check if it's type (F) - Aunt/Uncle
    % s152_d_2_relationship_qr checks all types. We need to confirm it's (F)
    % (F) A brother or sister of the father or mother of the taxpayer.
    % Taxpayer = Bob. Father of Bob = Charlie. Sister of Charlie = Alice. So Alice is sister of father of Bob.
    % This matches (F).
    ( RelationshipMet == true,
      ( fact(s152_d_2_F_pos, father_of(FatherBob, bob)), fact(s152_d_2_F_pos, sister_of(alice, FatherBob)) ) ;
      ( fact(s152_d_2_F_pos, mother_of(MotherBob, bob)), fact(s152_d_2_F_pos, sister_of(alice, MotherBob)) ) ;
      ( fact(s152_d_2_F_pos, father_of(FatherBob, bob)), fact(s152_d_2_F_pos, brother_of(alice, FatherBob)) ) ;
      ( fact(s152_d_2_F_pos, mother_of(MotherBob, bob)), fact(s152_d_2_F_pos, brother_of(alice, MotherBob)) )
    ) -> StatementTruth = true
    ; StatementTruth = false,
    fact(s152_d_2_F_pos, question_type(Type)),
    process_entailment_contradiction(StatementTruth, Type, Result).

    ( answer(CaseID, Result) ->
        format('Case ~w: Result = ~w~n', [CaseID, Result])
